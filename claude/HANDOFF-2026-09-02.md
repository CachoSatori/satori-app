```sql
DO $$
DECLARE
  v_emp_id        employees.id%TYPE;
  v_emp_count     int;
  v_antes         int;
  v_despues       int;
  v_otras_antes   int;
  v_otras_despues int;
BEGIN
  SELECT count(*) INTO v_emp_count FROM employees WHERE biotime_emp_code = '8';
  IF v_emp_count <> 1 THEN
    RAISE EXCEPTION 'Esperaba 1 empleado con biotime_emp_code=8, hay %', v_emp_count;
  END IF;
  SELECT id INTO v_emp_id FROM employees WHERE biotime_emp_code = '8';

  SELECT count(*) INTO v_antes       FROM time_punches WHERE emp_code =  '8' AND employee_id IS NULL;
  SELECT count(*) INTO v_otras_antes FROM time_punches WHERE emp_code <> '8' AND employee_id IS NULL;
  IF v_antes <> 1721 THEN
    RAISE EXCEPTION 'Huerfanas cod8 antes: esperaba 1721, hay %', v_antes;
  END IF;

  UPDATE time_punches
     SET employee_id = v_emp_id, updated_at = now()
   WHERE emp_code = '8' AND employee_id IS NULL;

  SELECT count(*) INTO v_despues       FROM time_punches WHERE emp_code =  '8' AND employee_id IS NULL;
  SELECT count(*) INTO v_otras_despues FROM time_punches WHERE emp_code <> '8' AND employee_id IS NULL;
  IF v_despues <> 0 THEN
    RAISE EXCEPTION 'Post: cod8 deberia quedar 0, quedo %', v_despues;
  END IF;
  IF v_otras_despues <> v_otras_antes THEN
    RAISE EXCEPTION 'Se tocaron otras marcas: antes % despues %', v_otras_antes, v_otras_despues;
  END IF;

  RAISE NOTICE 'OK: % marcas de Deivin enganchadas. Otras huerfanas intactas: %', v_antes, v_otras_despues;
END $$;

```
